Traversal and creation of array

C++:


Traversal of array: 

#include <iostream>
using namespace std;

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int n = 5;

  
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }

    return 0;
}


Insertion in Array:

1. Insert at First

#include <iostream>
using namespace std;

int main() {
    int arr[10] = {10, 20, 30, 40, 50};
    int n = 5;
    int element = 25;

    for(int i = n ; i > 0 ; i--){
        arr[i] = arr[i-1];
    }

    arr[0] = element ;
    n++;

    for(int i = 0 ; i < n ; i++){
        cout<<arr[i]<<"";
    }

    return 0;
}

2. Insert at last:

3. Insert at middle;

#include <iostream>
using namespace std;

int main() {
    int arr[] = {10, 20, 30, 40, 50};
    int n = 5;
    int element = 25;
    int pos = 2;

    for(int i = n ; i > pos ; i--){
        arr[i] = arr[i-1];
    }

    arr[pos] = element;
  
    for (int i = 0; i < n; i++) {
        cout << arr[i] << " ";
    }

    return 0;
}
